const express = require('express'); const request = require('request'); const cors = require('cors'); const app = express(); const port = 3000; // 允许所有跨域请求，彻底解决 CORS 问题 app.use(cors({ origin: "*", methods: ["GET", "OPTIONS"], allowedHeaders: ["api-key", "Content-Type"] })); // 处理预检请求 app.options('*', (req, res) => { res.sendStatus(200); }); // 代理 OneNet 请求 app.get('/api/onenet/devices/:deviceId/datapoints', (req, res) => { const { deviceId } = req.params; const apiKey = req.headers['api-key']; if (!apiKey) { return res.status(400).json({ error: 'Missing api-key header' }); } const oneNetUrl = `https://api.heclouds.com/devices/${deviceId}/datapoints`; // 转发请求到 OneNet request({ url: oneNetUrl, headers: { 'api-key': apiKey }, method: 'GET' }, (error, response, body) => { if (error) { console.error('OneNet request error:', error); return res.status(502).json({ error: 'Bad Gateway' }); } // 把 OneNet 的响应原样返回给前端 res.status(response.statusCode).send(body); }); }); app.listen(port, () => { console.log(`Proxy server running at http://localhost:${port}`); });
